# Daybook M1: Foundation and Month View Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A runnable iOS and macOS app that asks for calendar access, lists system calendars grouped by account with on/off toggles, and shows a month view in which every event of every enabled calendar is visible.

**Architecture:** Pure logic (models, store protocol, visibility filter, month layout, demo data) lives in the local Swift package `Packages/DaybookCore`, tested with `swift test`. The SwiftUI app target in `App/` adapts EventKit to the package's `CalendarStore` protocol and draws what `MonthLayout` computes. The Xcode project is generated by XcodeGen from `project.yml`.

**Tech Stack:** Swift (language mode 5), SwiftUI, EventKit, Swift Testing, XcodeGen, Xcode 26, iOS 26, macOS 26.

**Spec:** `docs/superpowers/specs/2026-11-17-daybook-design.md`. This plan covers build step 1 of section 9. Steps 2 to 5 get their own plans.

**Scope notes for M1:**
- One month is shown at a time, moved with back / Today / forward. Continuous scrolling between months (spec 4.3 rule 3) is deferred to the M5 plan.
- The Day / Week / Month switcher, search and "+ Event" arrive with the features they control (M2, M5, M3). M1 does not show dead controls.
- Categories (M4) are not in the sidebar yet.
- UI strings are written in Czech (development language `cs`). English is added in M5.

```swift
import Foundation

public enum CalendarAccess: Sendable, Equatable {
    case notDetermined, granted, denied
}

/// Everything the app needs from the system calendar database.
/// Implementations are used from the main actor only.
public protocol CalendarStore: AnyObject {
    var access: CalendarAccess { get }

    /// Asks the user for access if it has not been decided yet. Returns the resulting state.
    func requestAccess() async -> CalendarAccess

    /// All calendars the user has, switched on or not.
    func calendars() -> [CalendarInfo]

    /// Events that overlap `interval` and belong to one of `calendarIDs`, sorted by start then title.
    func events(in interval: DateInterval, calendarIDs: Set<String>) -> [EventInfo]

    /// Emits whenever calendars or events changed outside the app's own reads.
    var changes: AsyncStream<Void> { get }
}
```

```swift
import Foundation

/// A `CalendarStore` backed by arrays. Used by tests and by the app's demo mode.
public final class InMemoryCalendarStore: CalendarStore {
    public private(set) var access: CalendarAccess
    private var storedCalendars: [CalendarInfo]
    private var storedEvents: [EventInfo]
    private let continuation: AsyncStream<Void>.Continuation
    public let changes: AsyncStream<Void>

    public init(calendars: [CalendarInfo], events: [EventInfo], access: CalendarAccess = .granted) {
        self.storedCalendars = calendars
        self.storedEvents = events
        self.access = access
        let (stream, continuation) = AsyncStream<Void>.makeStream(bufferingPolicy: .bufferingNewest(1))
        self.changes = stream
        self.continuation = continuation
    }

    public func requestAccess() async -> CalendarAccess {
        if access == .notDetermined { access = .granted }
        return access
    }

    public func calendars() -> [CalendarInfo] { storedCalendars }

    public func events(in interval: DateInterval, calendarIDs: Set<String>) -> [EventInfo] {
        storedEvents
            .filter { calendarIDs.contains($0.calendarID) && $0.start < interval.end && $0.end > interval.start }
            .sorted { ($0.start, $0.title) < ($1.start, $1.title) }
    }

    public func replaceEvents(_ events: [EventInfo]) {
        storedEvents = events
        continuation.yield()
    }

    public func replaceCalendars(_ calendars: [CalendarInfo]) {
        storedCalendars = calendars
        continuation.yield()
    }
}
```

```swift
import Foundation

/// Remembers which calendars the user switched off. Everything else is visible,
/// so a calendar that appears later is shown without any action.
public struct VisibilityFilter: Equatable, Sendable {
    public private(set) var hiddenIDs: Set<String>

    public init(hiddenIDs: Set<String> = []) {
        self.hiddenIDs = hiddenIDs
    }

    /// Restores a filter from what `storedValue` produced.
    public init(storedValue: [String]) {
        self.hiddenIDs = Set(storedValue)
    }

    /// A stable, property-list friendly form for `UserDefaults`.
    public var storedValue: [String] { hiddenIDs.sorted() }

    public func isVisible(_ id: String) -> Bool { !hiddenIDs.contains(id) }

    public func visibleIDs(among all: Set<String>) -> Set<String> { all.subtracting(hiddenIDs) }

    public mutating func toggle(_ id: String) {
        if hiddenIDs.contains(id) { hiddenIDs.remove(id) } else { hiddenIDs.insert(id) }
    }

    public mutating func showOnly(_ id: String, among all: Set<String>) {
        hiddenIDs = all.subtracting([id])
    }

    public mutating func showAll() { hiddenIDs = [] }
}
```

```swift
import Foundation

public enum MonthLayout {
    /// The full weeks that cover the month containing `month`.
    /// Every date is the start of a day in `calendar`; weeks start on `calendar.firstWeekday`.
    public static func weeks(containing month: Date, calendar: Calendar) -> [[Date]] {
        guard let monthInterval = calendar.dateInterval(of: .month, for: month),
              var weekStart = calendar.dateInterval(of: .weekOfYear, for: monthInterval.start)?.start
        else { return [] }

        var result: [[Date]] = []
        while weekStart < monthInterval.end {
            result.append((0..<7).compactMap { calendar.date(byAdding: .day, value: $0, to: weekStart) })
            guard let next = calendar.date(byAdding: .day, value: 7, to: weekStart) else { break }
            weekStart = next
        }
        return result
    }
}
```

```swift
import Foundation

/// One day cell of the month grid.
public struct DayLayout: Hashable, Sendable {
    public let date: Date
    public let isInMonth: Bool
    /// Single-day timed events, sorted by start then title. Never truncated.
    public let timed: [EventInfo]
}

/// The part of an all-day or multi-day event that falls into one week.
public struct SpanSegment: Hashable, Sendable, Identifiable {
    public let event: EventInfo
    /// Vertical slot within the week, 0 at the top.
    public let lane: Int
    public let startColumn: Int
    /// Inclusive.
    public let endColumn: Int
    public let continuesBefore: Bool
    public let continuesAfter: Bool

    public var id: EventKey { event.key }
}

public struct WeekLayout: Hashable, Sendable, Identifiable {
    public let days: [DayLayout]
    public let spans: [SpanSegment]

    public var id: Date { days[0].date }
    public var laneCount: Int { (spans.map(\.lane).max() ?? -1) + 1 }
    /// Event rows the week needs so that its fullest day shows everything.
    public var rowCount: Int { laneCount + (days.map(\.timed.count).max() ?? 0) }
}

public enum MonthLayout {
    /// The full weeks that cover the month containing `month`.
    /// Every date is the start of a day in `calendar`; weeks start on `calendar.firstWeekday`.
    public static func weeks(containing month: Date, calendar: Calendar) -> [[Date]] {
        guard let monthInterval = calendar.dateInterval(of: .month, for: month),
              var weekStart = calendar.dateInterval(of: .weekOfYear, for: monthInterval.start)?.start
        else { return [] }

        var result: [[Date]] = []
        while weekStart < monthInterval.end {
            result.append((0..<7).compactMap { calendar.date(byAdding: .day, value: $0, to: weekStart) })
            guard let next = calendar.date(byAdding: .day, value: 7, to: weekStart) else { break }
            weekStart = next
        }
        return result
    }

    /// Places `events` into the weeks of the month containing `month`.
    public static func layout(month: Date, events: [EventInfo], calendar: Calendar) -> [WeekLayout] {
        let monthInterval = calendar.dateInterval(of: .month, for: month)
        let ranged = events.map { (event: $0, days: dayRange(of: $0, calendar: calendar)) }
        return weeks(containing: month, calendar: calendar).map { days in
            week(days: days, monthInterval: monthInterval, events: ranged)
        }
    }

    /// First and last day an event touches. An event ending exactly at midnight
    /// does not touch the day that starts at that midnight.
    static func dayRange(of event: EventInfo, calendar: Calendar) -> ClosedRange<Date> {
        let first = calendar.startOfDay(for: event.start)
        var lastInstant = event.end
        if event.end > event.start, calendar.startOfDay(for: event.end) == event.end {
            lastInstant = event.end.addingTimeInterval(-1)
        }
        return first...max(first, calendar.startOfDay(for: lastInstant))
    }

    private static func week(days: [Date], monthInterval: DateInterval?,
                             events: [(event: EventInfo, days: ClosedRange<Date>)]) -> WeekLayout {
        let firstDay = days[0]
        let lastDay = days[days.count - 1]
        var timed: [[EventInfo]] = Array(repeating: [], count: days.count)
        var spanning: [(event: EventInfo, days: ClosedRange<Date>)] = []

        for item in events {
            guard item.days.upperBound >= firstDay, item.days.lowerBound <= lastDay else { continue }
            if item.event.isAllDay || item.days.lowerBound != item.days.upperBound {
                spanning.append(item)
            } else if let column = days.firstIndex(of: item.days.lowerBound) {
                timed[column].append(item.event)
            }
        }

        // Earlier first, then longer, then by title, so lanes are stable and long events sit on top.
        spanning.sort { a, b in
            if a.days.lowerBound != b.days.lowerBound { return a.days.lowerBound < b.days.lowerBound }
            if a.days.upperBound != b.days.upperBound { return a.days.upperBound > b.days.upperBound }
            return a.event.title < b.event.title
        }

        var lastColumnOfLane: [Int] = []
        var spans: [SpanSegment] = []
        for item in spanning {
            let startColumn = days.firstIndex(of: max(item.days.lowerBound, firstDay)) ?? 0
            let endColumn = days.firstIndex(of: min(item.days.upperBound, lastDay)) ?? days.count - 1
            let lane = lastColumnOfLane.firstIndex { $0 < startColumn } ?? lastColumnOfLane.count
            if lane == lastColumnOfLane.count {
                lastColumnOfLane.append(endColumn)
            } else {
                lastColumnOfLane[lane] = endColumn
            }
            spans.append(SpanSegment(event: item.event, lane: lane,
                                     startColumn: startColumn, endColumn: endColumn,
                                     continuesBefore: item.days.lowerBound < firstDay,
                                     continuesAfter: item.days.upperBound > lastDay))
        }

        let dayLayouts = days.enumerated().map { index, date in
            DayLayout(date: date,
                      // DateInterval.contains includes the end instant, which is the next month's first midnight.
                      isInMonth: monthInterval.map { date >= $0.start && date < $0.end } ?? true,
                      timed: timed[index].sorted { ($0.start, $0.title) < ($1.start, $1.title) })
        }
        return WeekLayout(days: dayLayouts, spans: spans)
    }
}
```

Two heuristics live here because EventKit does not expose the facts directly:

**File:** `App/System/EventKitCalendarStore.swift`
```swift
import CoreGraphics
import DaybookCore
import EventKit
import Foundation

/// `CalendarStore` over the system calendar database.
final class EventKitCalendarStore: CalendarStore {
    private let store = EKEventStore()
    private let continuation: AsyncStream<Void>.Continuation
    private var observer: NSObjectProtocol?
    let changes: AsyncStream<Void>

    init() {
        let (stream, continuation) = AsyncStream<Void>.makeStream(bufferingPolicy: .bufferingNewest(1))
        self.changes = stream
        self.continuation = continuation
        observer = NotificationCenter.default.addObserver(
            forName: .EKEventStoreChanged, object: store, queue: .main
        ) { _ in continuation.yield() }
    }

    deinit {
        if let observer { NotificationCenter.default.removeObserver(observer) }
    }

    var access: CalendarAccess {
        switch EKEventStore.authorizationStatus(for: .event) {
        case .fullAccess: return .granted
        case .notDetermined: return .notDetermined
        default: return .denied
        }
    }

    func requestAccess() async -> CalendarAccess {
        guard access == .notDetermined else { return access }
        _ = try? await store.requestFullAccessToEvents()
        return access
    }

    func calendars() -> [CalendarInfo] {
        guard access == .granted else { return [] }
        return store.calendars(for: .event).map(Self.info(for:))
    }

    func events(in interval: DateInterval, calendarIDs: Set<String>) -> [EventInfo] {
        guard access == .granted else { return [] }
        let wanted = store.calendars(for: .event).filter { calendarIDs.contains($0.calendarIdentifier) }
        guard !wanted.isEmpty else { return [] }
        let predicate = store.predicateForEvents(withStart: interval.start, end: interval.end, calendars: wanted)
        return store.events(matching: predicate)
            .map(Self.info(for:))
            .sorted { ($0.start, $0.title) < ($1.start, $1.title) }
    }

    // MARK: - Translation

    private static func info(for calendar: EKCalendar) -> CalendarInfo {
        let kind = kind(of: calendar)
        let readOnly = !calendar.allowsContentModifications
        return CalendarInfo(id: calendar.calendarIdentifier,
                            title: calendar.title,
                            color: rgba(calendar.cgColor),
                            sourceTitle: calendar.source?.title ?? "",
                            sourceKind: kind,
                            isReadOnly: readOnly,
                            isShared: kind == .google && readOnly)
    }

    private static func kind(of calendar: EKCalendar) -> SourceKind {
        if calendar.type == .subscription { return .subscribed }
        guard let source = calendar.source else { return .other }
        let title = source.title.lowercased()
        switch source.sourceType {
        case .local: return .local
        case .exchange: return .exchange
        case .mobileMe: return .icloud
        case .subscribed: return .subscribed
        case .calDAV:
            if title.contains("icloud") { return .icloud }
            if title.contains("google") || title.contains("gmail") { return .google }
            return .other
        default: return .other
        }
    }

    private static func rgba(_ color: CGColor?) -> RGBA {
        guard let color,
              let space = CGColorSpace(name: CGColorSpace.sRGB),
              let converted = color.converted(to: space, intent: .defaultIntent, options: nil),
              let c = converted.components, c.count >= 3
        else { return RGBA(r: 0.56, g: 0.56, b: 0.58) }
        return RGBA(r: Double(c[0]), g: Double(c[1]), b: Double(c[2]), a: c.count > 3 ? Double(c[3]) : 1)
    }

    private static func info(for event: EKEvent) -> EventInfo {
        let start = event.startDate ?? Date()
        let externalID = event.calendarItemExternalIdentifier ?? event.eventIdentifier ?? UUID().uuidString
        return EventInfo(key: EventKey(externalID: externalID, occurrenceStart: event.occurrenceDate ?? start),
                         calendarID: event.calendar?.calendarIdentifier ?? "",
                         title: event.title ?? "",
                         start: start,
                         end: event.endDate ?? start,
                         isAllDay: event.isAllDay,
                         location: event.location,
                         notes: event.notes)
    }
}
```

```swift
/// Calendars of one account, as shown in the sidebar.
struct CalendarSection: Identifiable {
    let title: String
    let calendars: [CalendarInfo]
    var id: String { title }

/// A day the user picked, in a form `sheet(item:)` accepts.
struct SelectedDay: Identifiable {
    let date: Date
    var id: Date { date }
}

@MainActor
@Observable
final class AppModel {
    static let hiddenCalendarsKey = "hiddenCalendarIDs"

    let calendar: Calendar
    let today: Date
    private let store: CalendarStore
    private let defaults: UserDefaults
    private var watcher: Task<Void, Never>?

    private(set) var access: CalendarAccess
    private(set) var calendars: [CalendarInfo] = []
    private(set) var filter: VisibilityFilter
    /// Any instant inside the month on screen.
    private(set) var month: Date
    private(set) var weeks: [WeekLayout] = []

    init(store: CalendarStore, defaults: UserDefaults = .standard,
         calendar: Calendar = .current, now: Date = Date()) {
        self.store = store
        self.defaults = defaults
        self.calendar = calendar
        self.today = calendar.startOfDay(for: now)
        self.month = now
        self.access = store.access
        self.filter = VisibilityFilter(storedValue: defaults.stringArray(forKey: Self.hiddenCalendarsKey) ?? [])
    }

    /// Loads the first month and keeps reloading when the system database changes.
    func start() {
        reload()
        guard watcher == nil else { return }
        watcher = Task { [weak self] in
            guard let changes = self?.store.changes else { return }
            for await _ in changes { self?.reload() }
        }
    }

    func requestAccess() async {
        access = await store.requestAccess()
        reload()
    }

    func reload() {
        access = store.access
        calendars = access == .granted ? store.calendars() : []
        let days = MonthLayout.weeks(containing: month, calendar: calendar).flatMap { $0 }
        var events: [EventInfo] = []
        if access == .granted, let first = days.first, let last = days.last,
           let end = calendar.date(byAdding: .day, value: 1, to: last) {
            let visible = filter.visibleIDs(among: Set(calendars.map(\.id)))
            events = store.events(in: DateInterval(start: first, end: end), calendarIDs: visible)
        }
        weeks = MonthLayout.layout(month: month, events: events, calendar: calendar)
    }

    // MARK: - Calendars

    private static let kindOrder: [SourceKind] = [.icloud, .exchange, .google, .local, .subscribed, .other]

    var sections: [CalendarSection] {
        Dictionary(grouping: calendars, by: \.sourceTitle)
            .map { title, items in
                CalendarSection(title: title.isEmpty ? "Ostatní" : title,
                                calendars: items.sorted { $0.title.localizedStandardCompare($1.title) == .orderedAscending })
            }
            .sorted { a, b in
                let ka = Self.kindOrder.firstIndex(of: a.calendars[0].sourceKind) ?? Self.kindOrder.count
                let kb = Self.kindOrder.firstIndex(of: b.calendars[0].sourceKind) ?? Self.kindOrder.count
                return (ka, a.title) < (kb, b.title)
            }
    }

    func calendarInfo(_ id: String) -> CalendarInfo? { calendars.first { $0.id == id } }

    func toggle(_ calendar: CalendarInfo) { updateFilter { $0.toggle(calendar.id) } }

    func showOnly(_ calendar: CalendarInfo) {
        let all = Set(calendars.map(\.id))
        updateFilter { $0.showOnly(calendar.id, among: all) }
    }

    func showAll() { updateFilter { $0.showAll() } }

    private func updateFilter(_ change: (inout VisibilityFilter) -> Void) {
        change(&filter)
        defaults.set(filter.storedValue, forKey: Self.hiddenCalendarsKey)
        reload()
    }

    // MARK: - Navigation

    func step(months: Int) {
        guard let start = calendar.dateInterval(of: .month, for: month)?.start,
              let moved = calendar.date(byAdding: .month, value: months, to: start) else { return }
        month = moved
        reload()
    }

    func goToToday() {
        month = today
        reload()
    }

    /// Everything planned on `day`, all-day and multi-day events first.
    func events(on day: Date) -> [EventInfo] {
        for week in weeks {
            guard let column = week.days.firstIndex(where: { $0.date == day }) else { continue }
            let spans = week.spans
                .filter { $0.startColumn <= column && column <= $0.endColumn }
                .sorted { $0.lane < $1.lane }
                .map(\.event)
            return spans + week.days[column].timed
        }
        return []
    }
}
```

```swift
        .overlay(alignment: .bottom) { Rectangle().fill(Theme.rule).frame(height: 1) }

    private func dayCell(_ day: DayLayout, isLast: Bool) -> some View {
        VStack(alignment: .leading, spacing: 0) {
            dayNumber(day)
            Color.clear.frame(height: CGFloat(week.laneCount) * rowHeight)
            ForEach(day.timed) { event in
                EventRowView(event: event, color: color(event.calendarID), compact: compact)
                    .frame(height: rowHeight)
            }
            Spacer(minLength: 6)
        }
        .frame(maxWidth: .infinity, maxHeight: .infinity, alignment: .topLeading)
        .background(calendar.isDateInWeekend(day.date) ? Theme.weekend : Color.clear)
        .opacity(day.isInMonth ? 1 : 0.45)
        .clipped()
        .overlay(alignment: .trailing) {
            if !isLast { Rectangle().fill(Theme.rule).frame(width: 1) }
        }
        .contentShape(Rectangle())
        .onTapGesture { onSelectDay?(day.date) }
        .accessibilityElement(children: .contain)
        .accessibilityLabel(day.date.formatted(date: .complete, time: .omitted))
    }

    private func dayNumber(_ day: DayLayout) -> some View {
        let isToday = day.date == today
        return Text(day.date, format: .dateTime.day())
            .font(.callout.weight(.semibold))
            .monospacedDigit()
            .foregroundStyle(isToday ? Theme.ground : Theme.ink)
            .padding(.horizontal, isToday ? 6 : 0)
            .padding(.vertical, 2)
            .background { if isToday { Capsule().fill(Theme.ink) } }
            .padding(.leading, compact ? 3 : 6)
            .frame(height: numberHeight, alignment: .leading)
            .accessibilityLabel(isToday ? "Dnes" : "")
    }

    private var spans: some View {
        GeometryReader { proxy in
            let column = proxy.size.width / CGFloat(week.days.count)
            ZStack(alignment: .topLeading) {
                ForEach(week.spans) { span in
                    SpanView(span: span, color: color(span.event.calendarID))
                        .frame(width: column * CGFloat(span.endColumn - span.startColumn + 1), height: rowHeight)
                        .offset(x: column * CGFloat(span.startColumn),
                                y: numberHeight + CGFloat(span.lane) * rowHeight)
                }
            }
        }
        .allowsHitTesting(false)
    }
}
```

```swift
import DaybookCore
import SwiftUI

/// The access gate, and behind it the split view (Mac, iPad) or single column (iPhone).
struct RootView: View {
    @Environment(AppModel.self) private var model
    @Environment(\.horizontalSizeClass) private var sizeClass
    @State private var showsCalendars = false

    var body: some View {
        if model.access == .granted {

        } else {
            AccessView(access: model.access) {
                Task { await model.requestAccess() }
            }
        }
    }

    @ViewBuilder private var shell: some View {
        if sizeClass == .compact {
            NavigationStack {
                month.toolbar {
                    ToolbarItem(placement: .navigation) {
                        Button("Kalendáře", systemImage: "sidebar.leading") { showsCalendars = true }
                    }
                }
            }
            .sheet(isPresented: $showsCalendars) {
                NavigationStack {
                    SidebarView()
                        .navigationTitle("Kalendáře")
                        .toolbar {
                            ToolbarItem(placement: .confirmationAction) {
                                Button("Hotovo") { showsCalendars = false }
                            }
                        }
                }
                .presentationDetents([.medium, .large])
            }
        } else {
            NavigationSplitView {
                SidebarView()
                    .navigationTitle("Daybook")
                    .navigationSplitViewColumnWidth(min: 220, ideal: 248, max: 320)
            } detail: {
                month
            }
        }
    }

    private var month: some View {
        MonthView()
            .navigationTitle(model.month.formatted(.dateTime.month(.wide).year()).localizedCapitalized)
            #if os(iOS)
            .navigationBarTitleDisplayMode(.inline)
            #endif
            .toolbar {
                ToolbarItemGroup(placement: .primaryAction) {
                    Button("Předchozí měsíc", systemImage: "chevron.left") { model.step(months: -1) }
                        .keyboardShortcut(.leftArrow, modifiers: .command)
                    Button("Dnes") { model.goToToday() }
                        .keyboardShortcut("t", modifiers: .command)
                    Button("Další měsíc", systemImage: "chevron.right") { model.step(months: 1) }
                        .keyboardShortcut(.rightArrow, modifiers: .command)
                }
            }
    }
```
